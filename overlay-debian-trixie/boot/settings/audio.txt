## Audio: where the OpenFrame's sound goes.

# output=builtin for its own speaker and headphone socket, or a USB DAC by its USB id
# (vendor:product, such as 08bb:2704: Sprite's Settings show it, under Audio). A DAC plugged
# in for the first time becomes the output by itself; unplugged, the sound comes from the
# OpenFrame until it's back.
output=builtin

# The DACs that get the audio direct, by their USB ids (spaces between): each is sent the
# audio as it is, with nothing mixed in or resampled and its own volume at full, so set the
# volume on your amplifier; only one app can play at a time. A DAC not listed lets the
# OpenFrame set the volume, and mixes the sound of several apps.
direct=

# The rate the OpenFrame mixes at when the audio isn't sent direct. 44100 (44.1 kHz) suits
# music, most of which is made at that rate, and 48000 (48 kHz) suits video. Audio at the
# other rate is converted cleanly.
mixer_rate=44100
