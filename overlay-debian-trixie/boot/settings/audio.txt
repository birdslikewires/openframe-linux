## Audio: where the OpenFrame's sound goes.

# output=builtin for its own speaker and headphone socket, or a USB DAC by its USB id
# (vendor:product, such as 08bb:2704: Sprite's Settings show it, under Audio). A DAC plugged
# in for the first time becomes the output by itself; unplugged, the sound comes from the
# OpenFrame until it's back.
output=builtin

# For a DAC: direct=yes sends it the audio as it is, with nothing mixed in or resampled and
# its own volume at full: set the volume on your amplifier. Only one app can play at a time.
# direct=no lets the OpenFrame set the volume, and mixes the sound of several apps.
direct=no
